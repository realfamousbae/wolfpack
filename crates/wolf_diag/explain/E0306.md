# E0306: struct literal in a condition without parentheses

In the condition of `if`, `while`, `for` and `match` a `{` opens the body, so a struct literal there
is ambiguous. Wrap the literal in parentheses.

Wrong:

```wolf wrong
module main

types {
    struct Point {
        x: Int
        y: Int
    }
}

main {
    print(is_origin(Point { x: 0, y: 0 }).to_string())
}

functions {
    fn is_origin(p: Point) -> Bool {
        if p == Point { x: 0, y: 0 } {
            return true
        }
        false
    }
}
```

Fixed:

```wolf fixed
module main

types {
    struct Point {
        x: Int
        y: Int
    }
}

main {
    print(is_origin(Point { x: 0, y: 0 }).to_string())
}

functions {
    fn is_origin(p: Point) -> Bool {
        if p == (Point { x: 0, y: 0 }) {
            return true
        }
        false
    }
}
```
