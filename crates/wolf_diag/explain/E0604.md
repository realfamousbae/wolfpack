# E0604: import cycle

Modules import each other in a cycle. Imports must form a tree, so that every target can build the
modules in order (Go does not allow import cycles). Move the declarations that both need into a
third module that both import.

Wrong:

```wolf wrong
module main

imports {
    a.{one}
}

functions {
    fn main() {
        print(one().to_string())
    }
}

// file: src/a.wolf
module a

imports {
    b.{two}
}

functions {
    pub fn one() -> Int {
        two() - 1
    }
}

// file: src/b.wolf
module b

imports {
    a.{one}
}

functions {
    pub fn two() -> Int {
        2
    }

    pub fn three() -> Int {
        one() + 2
    }
}
```

Fixed:

```wolf fixed
module main

imports {
    a.{one}
}

functions {
    fn main() {
        print(one().to_string())
    }
}

// file: src/a.wolf
module a

imports {
    b.{two}
}

functions {
    pub fn one() -> Int {
        two() - 1
    }

    pub fn three() -> Int {
        one() + 2
    }
}

// file: src/b.wolf
module b

functions {
    pub fn two() -> Int {
        2
    }
}
```
